2.4 Prompt Caching & Structured Outputs

xAI API natively supports Automatic KV Prompt Caching on prompts exceeding 1,024 tokens. Cached tokens receive an automatic 75% cost reduction and up to 80% TTFT latency reduction.

import os
from openai import OpenAI
from pydantic import BaseModel, Field

client = OpenAI(
    api_key=os.environ.get("XAI_API_KEY"),
    base_url="https://api.x.ai/v1",
)

class SecurityAudit(BaseModel):
    is_safe: bool
    vulnerabilities: list[str] = Field(description="List of detected CVEs or code flaws")
    recommendation: str
    refactored_code: str

completion = client.beta.chat.completions.parse(
    model="grok-3",
    messages=[
        {"role": "system", "content": "You are Grok, an elite application security engineer."},
        {"role": "user", "content": "Audit this snippet: os.system('cat ' + filename)"}
    ],
    response_format=SecurityAudit,
)

print(completion.choices[0].message.parsed)

3.5 Project: Automated PR Auditor Bot

import os, requests
from openai import OpenAI

def run_pr_audit():
    gh_token = os.environ.get("GITHUB_TOKEN")
    repo = os.environ.get("GITHUB_REPOSITORY")
    pr_num = os.environ.get("PR_NUMBER")
    
    diff_resp = requests.get(
        f"https://api.github.com/repos/{repo}/pulls/{pr_num}",
        headers={"Authorization": f"token {gh_token}", "Accept": "application/vnd.github.v3.diff"}
    )
    
    client = OpenAI(api_key=os.environ.get("XAI_API_KEY"), base_url="https://api.x.ai/v1")
    
    res = client.chat.completions.create(
        model="grok-3",
        messages=[
            {"role": "system", "content": "You are Grok, an elite Code Security Auditor."},
            {"role": "user", "content": f"Review this PR diff for security, memory leaks, and performance:\n\n{diff_resp.text[:15000]}"}
        ]
    )
    
    requests.post(
        f"https://api.github.com/repos/{repo}/issues/{pr_num}/comments",
        headers={"Authorization": f"token {gh_token}", "Accept": "application/vnd.github.v3+json"},
        json={"body": f"### 🤖 Grok-3 Automated Code Review\n\n{res.choices[0].message.content}"}
    )
    print("PR review posted.")

if __name__ == "__main__":
    run_pr_audit()
